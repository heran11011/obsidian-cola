# Obsidian-Cola 插件技术方案

更新时间：2026-06-06

---

## 1. 项目概述

### 1.1 目标

在 Obsidian 侧边栏内嵌一个 Cola 聊天面板，用户无需切换窗口即可与 Cola 对话。Cola 能自动感知当前打开的文件，实现深度联动。

### 1.2 产物

| 组件 | 说明 | 发布渠道 |
|------|------|----------|
| `obsidian-cola` | Obsidian 社区插件，提供侧边栏 UI + 文件感知 + 与 Cola 通信 | Obsidian 社区插件市场 |
| `cola-plugin-obsidian` | Cola Channel Plugin，接收 Obsidian 端消息并回复 | Cola 插件商店 |

### 1.3 用户体验

```
用户在 Obsidian 侧边栏输入消息
    ↓ 自动附带当前文件路径+内容
Obsidian 插件 → Cola Channel Plugin（本地通信）
    ↓
Cola 处理（完整 Agent 能力）
    ↓
回复推回 Obsidian 侧边栏渲染
```

### 2.1 整体架构

```
│  │  │  (WebSocket 客户端)          │ │  │
│  │  └──────────────┬───────────────┘ │  │

│  ┌─────────────────┴──────────────────┐ │
│  │   cola-plugin-obsidian             │ │
│  │   (Channel Plugin)                 │ │
│  └─────────────────┬──────────────────┘ │
│                    │                     │
│  Cola Agent 处理 ← ┘                    │
└──────────────────────────────────────────┘
```

### 2.2 关键架构决策

**决策 1：通信方式选择**

| 方案 | 优点 | 缺点 |
|------|------|------|
| A. Obsidian 直连 Cola WebSocket RPC | 简单、一步到位 | 绕过 Channel Plugin 体系，Cola 端无法在插件商店管理 |
| B. Cola Channel Plugin + Obsidian 插件配合 | 标准化、可管理、可扩展 | 多一层，但架构更正 |

**选择方案 B**。理由：
- 走 Channel Plugin 体系，Cola 端有正式的插件管理入口
- 状态可查（`/obsidian status`）
- 未来可扩展（多 vault、远程 vault）
- 符合 Cola 生态规范

**决策 2：Obsidian 插件如何连接 Cola Channel Plugin**

Cola Channel Plugin 的 gateway.start 中启动一个本地 HTTP/WebSocket 服务（比如 `127.0.0.1:19533`），Obsidian 插件作为客户端连接它。

为什么不让 Obsidian 直接连 Cola 的 19532 端口？因为 19532 是 Cola 内部 RPC 接口，Cola Channel Plugin SDK 提供的 `ctx.deliver()` 才是正确的消息入口。Channel Plugin 自己起一个端口给 Obsidian 连，收到消息后调 `ctx.deliver()`。

```
Obsidian 插件 ──ws://127.0.0.1:19533──→ Cola Channel Plugin
                                            │
                                            ↓ ctx.deliver()
                                         Cola Agent
                                            │
                                            ↓ outbound.sendText()
Cola Channel Plugin ──ws push──→ Obsidian 插件
```

**决策 3：认证方式**

自动发现，零配置：
1. Cola Channel Plugin 启动时生成一个本地 token，写入 `~/.cola/plugins/obsidian/local-token`
2. Obsidian 插件启动时读取该文件（Obsidian 在本机，有文件读取权限）
3. 连接时附带 token，完成认证

用户不需要手动填任何东西。

```typescript
import { Plugin } from 'obsidian';
import { ColaView, VIEW_TYPE_COLA } from './views/ColaView';
import { ColaGateway } from './services/ColaGateway';
import { ColaSettingTab } from './settings/SettingTab';

export default class ColaPlugin extends Plugin {
  gateway: ColaGateway;

  async onload() {
    this.gateway = new ColaGateway(this);

    // 注册侧边栏视图
    this.registerView(VIEW_TYPE_COLA, (leaf) => new ColaView(leaf, this));

    // 功能区图标
    this.addRibbonIcon('message-circle', 'Open Cola', () => {
      this.activateView();
    });

    // 命令
    this.addCommand({
      id: 'open-cola-chat',
      name: 'Open Cola Chat',
      callback: () => this.activateView(),
    });

    // 连接 Cola
    await this.gateway.connect();
  }

  async onunload() {
    this.gateway.disconnect();
  }

  async activateView() {
    const { workspace } = this.app;
    const leaves = workspace.getLeavesOfType(VIEW_TYPE_COLA);

    if (leaves.length > 0) {
      workspace.revealLeaf(leaves[0]);
    } else {
      const leaf = workspace.getRightLeaf(false);
      await leaf.setViewState({ type: VIEW_TYPE_COLA, active: true });
      workspace.revealLeaf(leaf);
    }
  }
}
```

**views/ColaView.ts**
```typescript
import { ItemView, WorkspaceLeaf, TFile } from 'obsidian';
import type ColaPlugin from '../main';

export const VIEW_TYPE_COLA = 'cola-chat-view';

interface ChatMessage {
  role: 'user' | 'assistant';
  content: string;
  timestamp: number;
}

export class ColaView extends ItemView {
  plugin: ColaPlugin;
  messages: ChatMessage[] = [];
  chatContainer: HTMLElement;
  inputEl: HTMLTextAreaElement;

  constructor(leaf: WorkspaceLeaf, plugin: ColaPlugin) {
    super(leaf);
    this.plugin = plugin;
  }

  getViewType(): string { return VIEW_TYPE_COLA; }
  getDisplayText(): string { return 'Cola'; }
  getIcon(): string { return 'message-circle'; }

  async onOpen() {
    const container = this.contentEl;
    container.empty();
    container.addClass('cola-chat-container');

    // 状态栏：显示当前文件
    const statusBar = container.createEl('div', { cls: 'cola-status-bar' });
    this.updateStatusBar(statusBar);

    // 聊天消息区
    this.chatContainer = container.createEl('div', { cls: 'cola-messages' });

    // 输入区
    const inputArea = container.createEl('div', { cls: 'cola-input-area' });
    this.inputEl = inputArea.createEl('textarea', {
      attr: { placeholder: '和 Cola 说点什么...', rows: '3' }
    });
    const sendBtn = inputArea.createEl('button', { text: '发送', cls: 'cola-send-btn' });

    // 发送事件
    sendBtn.addEventListener('click', () => this.sendMessage());
    this.inputEl.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
        e.preventDefault();
        this.sendMessage();
      }
    });

    // 监听文件切换
    this.registerEvent(
      this.app.workspace.on('file-open', () => this.updateStatusBar(statusBar))
    );

    // 监听 Cola 回复
    this.plugin.gateway.onMessage((text) => {
      this.addMessage('assistant', text);
    });
  }

  async onClose() {
    // 自动清理 registerEvent
  }

  private updateStatusBar(el: HTMLElement) {
    const file = this.app.workspace.getActiveFile();
    el.setText(file ? `📄 ${file.basename}` : '无打开文件');
  }

  private async sendMessage() {
    const text = this.inputEl.value.trim();
    if (!text) return;

    this.inputEl.value = '';
    this.addMessage('user', text);

    // 获取当前文件上下文
    const context = await this.getFileContext();

    // 发送给 Cola
    this.plugin.gateway.send({
      text,
      context,
    });
  }

  private async getFileContext(): Promise<object | null> {
    const file = this.app.workspace.getActiveFile();
    if (!file) return null;

    const content = await this.app.vault.cachedRead(file);
    return {
      filePath: file.path,
      fileName: file.basename,
      content: content.slice(0, 10000), // 限制长度，避免超大文件
    };
  }

  private addMessage(role: 'user' | 'assistant', content: string) {
    this.messages.push({ role, content, timestamp: Date.now() });
    const msgEl = this.chatContainer.createEl('div', {
      cls: `cola-msg cola-msg-${role}`,
    });
    msgEl.setText(content); // MVP 先用纯文本，后续支持 Markdown 渲染
    this.chatContainer.scrollTop = this.chatContainer.scrollHeight;
  }
}
```

**services/ColaGateway.ts**
```typescript
import { readFileSync } from 'fs';
import { join } from 'path';
import { homedir } from 'os';

interface ColaMessage {
  text: string;
  context?: object | null;
}

export class ColaGateway {
  private ws: WebSocket | null = null;
  private messageHandler: ((text: string) => void) | null = null;
  private reconnectTimer: NodeJS.Timeout | null = null;
  private port = 19533; // Cola Obsidian Channel Plugin 的端口

  constructor(private plugin: any) {}

  async connect() {
    try {
      const token = this.readToken();
      const url = `ws://127.0.0.1:${this.port}?token=${token}`;
      this.ws = new WebSocket(url);

      this.ws.onopen = () => {
        console.log('[Cola] Connected');
      };

      this.ws.onmessage = (event) => {
        const data = JSON.parse(event.data);
        if (data.type === 'reply' && this.messageHandler) {
          this.messageHandler(data.text);
        }
      };

      this.ws.onclose = () => {
        console.log('[Cola] Disconnected, will retry...');
        this.scheduleReconnect();
      };

      this.ws.onerror = (err) => {
        console.error('[Cola] WebSocket error:', err);
      };
    } catch (e) {
      console.error('[Cola] Failed to connect:', e);
      this.scheduleReconnect();
    }
  }

  disconnect() {
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
    if (this.ws) this.ws.close();
  }

  send(msg: ColaMessage) {
    if (!this.ws || this.ws.readyState !== WebSocket.OPEN) {
      console.error('[Cola] Not connected');
      return;
    }
    this.ws.send(JSON.stringify({
      type: 'message',
      text: msg.text,
      context: msg.context,
    }));
  }

  onMessage(handler: (text: string) => void) {
    this.messageHandler = handler;
  }

  private readToken(): string {
    const tokenPath = join(homedir(), '.cola', 'plugins', 'obsidian', 'local-token');
    return readFileSync(tokenPath, 'utf-8').trim();
  }

  private scheduleReconnect() {
    if (this.reconnectTimer) return;
    this.reconnectTimer = setTimeout(() => {
      this.reconnectTimer = null;
      this.connect();
    }, 5000);
  }
}
```

**src/index.ts**
```typescript
import { defineChannel } from "@marswave/cola-plugin-sdk";
import type { ChannelStatusResult, OutboundContext } from "@marswave/cola-plugin-sdk";
import { startLocalServer, stopLocalServer, getServerStatus } from "./gateway.js";
import { sendReplyToObsidian } from "./outbound.js";

interface ObsidianState {
  server: any;
  connections: Map<string, WebSocket>;
}

export default defineChannel<ObsidianState>({
  id: "obsidian",

  meta: {
    label: "Obsidian",
    description: "Chat with Cola inside Obsidian with full vault context",
    markdownCapable: true,
  },

  capabilities: {
    receive: { text: true },
    send: { text: true, markdown: true },
    limits: { maxTextLength: 50000 },
  },

  config: {
    schema: {
      fields: [
        {
          key: "port",
          label: "Local port",
          type: "text",
          required: false,
          placeholder: "19533",
          defaultValue: "19533",
        },
      ],
    },
  },

  gateway: {
    async start(ctx) {
      ctx.state.connections = new Map();
      ctx.state.server = await startLocalServer(ctx);
      ctx.logger.info("Obsidian gateway started");
    },

    async stop(ctx) {
      await stopLocalServer(ctx);
      ctx.state.connections.clear();
      ctx.logger.info("Obsidian gateway stopped");
    },

    async reload(ctx) {
      await stopLocalServer(ctx);
      ctx.state.connections = new Map();
      ctx.state.server = await startLocalServer(ctx);
    },

    getStatus(ctx): ChannelStatusResult {
      return getServerStatus(ctx);
    },
  },

  outbound: {
    async sendText(ctx: OutboundContext) {
      await sendReplyToObsidian(ctx);
    },
  },
});
```

**src/gateway.ts**（核心：本地 WebSocket Server）
```typescript
import { WebSocketServer } from "ws";
import { randomBytes, writeFileSync } from "fs";
import { join } from "path";
import { homedir } from "os";
import { mkdirSync } from "fs";

export async function startLocalServer(ctx: any) {
  const port = parseInt(ctx.config?.port || "19533");
  const token = randomBytes(32).toString("hex");

  // 写 token 文件供 Obsidian 插件读取
  const tokenDir = join(homedir(), ".cola", "plugins", "obsidian");
  mkdirSync(tokenDir, { recursive: true });
  writeFileSync(join(tokenDir, "local-token"), token, { mode: 0o600 });

  const wss = new WebSocketServer({ port, host: "127.0.0.1" });

  wss.on("connection", (ws, req) => {
    // 验证 token
    const url = new URL(req.url!, `http://127.0.0.1:${port}`);
    if (url.searchParams.get("token") !== token) {
      ws.close(4001, "Unauthorized");
      return;
    }

    const connId = randomBytes(8).toString("hex");
    ctx.state.connections.set(connId, ws);
    ctx.logger.info(`Obsidian client connected: ${connId}`);

    ws.on("message", async (raw: Buffer) => {
      try {
        const data = JSON.parse(raw.toString());
        if (data.type === "message") {
          // 构造消息文本：如果有文件上下文，附加到消息
          let text = data.text;
          if (data.context) {
            text = `[当前文件: ${data.context.filePath}]\n\n---文件内容---\n${data.context.content}\n---文件内容结束---\n\n用户问题: ${data.text}`;
          }

          // 递交给 Cola
          await ctx.deliver({
            from: `obsidian:${connId}`,
            to: "obsidian",
            text,
          });
        }
      } catch (e) {
        ctx.logger.error("Failed to process Obsidian message", e);
      }
    });

    ws.on("close", () => {
      ctx.state.connections.delete(connId);
      ctx.logger.info(`Obsidian client disconnected: ${connId}`);
    });
  });

  return wss;
}

export async function stopLocalServer(ctx: any) {
  if (ctx.state.server) {
    ctx.state.server.close();
    ctx.state.server = null;
  }
}

export function getServerStatus(ctx: any) {
  if (!ctx.state.server) {
    return { connected: false, configured: true, message: "Server not running" };
  }
  const count = ctx.state.connections?.size || 0;
  return {
    connected: count > 0,
    configured: true,
    message: count > 0 ? `${count} Obsidian client(s) connected` : "Waiting for Obsidian connection",
  };
}
```

**src/outbound.ts**
```typescript
import type { OutboundContext } from "@marswave/cola-plugin-sdk";

export async function sendReplyToObsidian(ctx: OutboundContext) {
  // 从 deliveryContext.to 解析出连接 ID
  const connId = ctx.deliveryContext.to.replace("obsidian:", "");
  const ws = (ctx as any).state?.connections?.get(connId);

  if (!ws || ws.readyState !== 1) {
    ctx.logger.error("sendText: Obsidian client not connected");
    return;
  }

  ws.send(JSON.stringify({
    type: "reply",
    text: ctx.text,
  }));
}
```
